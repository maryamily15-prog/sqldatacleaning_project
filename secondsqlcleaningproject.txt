--Checking inconsistent text data displaying string function in sql
 SELECT
  customer_name,
    TRIM(customer_name) AS fixed_customer_name,
  email,
    TRIM(email) AS email,
  city,
  TRIM(city) AS city,
  country,
  TRIM(country) AS country,
  Product_name,
  TRIM (product_name) AS product_name,
  category,
  TRIM(category) AS category
  FROM PROJECT2.dbo.sql_cleaning

  SELECT 
  country AS Original_country,
  CASE
  WHEN UPPER(TRIM(country)) IN ('NG','NIGERIA') THEN 'Nigeria'
  WHEN UPPER(TRIM(country)) IN ('CA','CANADA') THEN 'Canada'
  WHEN UPPER(TRIM(country)) IN ('US','USA' ,'UNITED STATES') THEN 'United States'
  WHEN UPPER(TRIM(country)) IN ('UK','UNITED KINGDOM') THEN 'United kingdom'
  WHEN UPPER(TRIM(country)) = 'GHANA' THEN 'Ghana'
  WHEN UPPER(TRIM(country)) = 'AUSTRALIA' THEN 'Australia'
  WHEN UPPER(TRIM(country)) = 'SOUTH AFRICA' THEN 'South Africa'
  WHEN UPPER(TRIM(country)) = 'KENYA' THEN 'Kenya'
  ELSE 'Country'
  END AS Clean_country
  FROM PROJECT2.dbo.sql_cleaning

  SELECT 
  Gender,
  CASE
  WHEN UPPER(TRIM(Gender)) IN ('F', 'f','FEMALE') THEN 'Female'
  WHEN UPPER(TRIM(Gender)) IN ('M', 'm','MALE') THEN 'Male'
  ELSE 'NULL'
  END AS CLEAN
  FROM PROJECT2.dbo.sql_cleaning

SELECT 
order_id,
customer_name,
phone,
LEN(
   REPLACE(phone,'+','')
   ) AS Phone_length
FROM sql_cleaning_worktable
WHERE phone IS NOT NULL
AND phone <> ''
AND (
    LEN(REPLACE(phone, '+', '')) < 7
OR LEN(REPLACE(phone, '+', '')) > 15
OR phone LIKE '%[^0-9+]%'
)
ORDER BY Phone_length DESC

 quantity,
 unit_price,
 total_amount,

 ABS(CAST(total_amount AS DECIMAL(18, 2)) - (CAST(quantity AS DECIMAL(18, 2))

 *CAST(unit_price AS DECIMAL(18, 2)) )
 ) AS differences
 FROM sql_cleaning_worktable

 WHERE quantity IS NOT NULL
 AND unit_price IS NOT NULL
 AND ABS(
        CAST(total_amount AS DECIMAL(18, 2))
        -(
        CAST(total_amount AS DECIMAL(18, 2))
        *CAST(unit_price AS DECIMAL(18, 2))
        )
        ) > 0.01
       ORDER BY differences DESC;
     


--seperating our data from good and bad
 ALTER TABLE PROJECT2.DBO.sql_cleaning_worktable
 ADD data_quality_flag VARCHAR(100) NULL;

 UPDATE sql_cleaning_worktable
 SET data_quality_flag =
     CASE 
     WHEN quantity IS NULL
         THEN 'MISSING QUANTITY'
         WHEN quantity <= 0
           THEN 'INVALID QUANTITY'
           WHEN quantity > 100
             THEN 'HIGH QUANTITY'
            WHEN unit_price IS NULL
             THEN  'MISSING UNIT PRICE'
            WHEN unit_price <= 0
             THEN 'INVALID UNIT PRICE'
            WHEN total_amount IS NULL
             THEN 'MISSING TOTAL'
            WHEN total_amount < 0
             THEN 'NEGATIVE TOTAL'
            WHEN total_amount > 10000
             THEN 'HIGH TOTAL'
            WHEN quantity IS NOT NULL
             AND unit_price IS NOT NULL
             AND total_amount IS NOT NULL
             AND ABS(total_amount - (quantity * unit_price)) > 0.01
             THEN 'TOTAL MISMATCH'
             ELSE 'PASS'
             END;
